raw_manifest <- utils::read.csv(
  here::here("data", "raw", "manifest.csv"),
  stringsAsFactors = FALSE,
  check.names = FALSE
)
source_inventory <- aggregate(
  cbind(files = rep(1L, nrow(raw_manifest)), bytes = raw_manifest$bytes),
  by = list(provider = raw_manifest$provider),
  FUN = sum
)
source_inventory$size_mb <- round(source_inventory$bytes / 1024^2, 1)
